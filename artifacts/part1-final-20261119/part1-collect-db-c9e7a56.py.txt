import datetime, json, subprocess, time
from pathlib import Path
sql = """SELECT json_build_object(
 'runs',(SELECT json_agg(r) FROM (SELECT project,principal,bot,run_id,thread_id,status,fence FROM hermes_agui_host.runs ORDER BY created_at) r),
 'profiles',(SELECT json_agg(r) FROM (SELECT project,principal,bot,profile->>'revision' AS revision FROM hermes_agui_host.bot_state) r),
 'managed',(SELECT json_agg(r) FROM (SELECT project,principal,bot,status,applied_revision FROM hermes_agui_host.managed_bots) r),
 'mappings',(SELECT json_agg(r) FROM (SELECT project,principal,bot,browser_thread,host_thread FROM hermes_agui_runtime.thread_mappings) r),
 'jobs',(SELECT json_agg(r) FROM (SELECT id,ns,source,state,attempt FROM hermes_agui_runtime.graph_jobs) r),
 'sources',(SELECT json_agg(r) FROM (SELECT ns,id,version,bot,deleted FROM hermes_agui_runtime.graph_sources) r)
)"""
base = Path('/private/tmp/part1-final-restart-c9e7a56')
ready_deadline = time.monotonic() + 240
while not list(base.rglob('*-host.inspect.txt')):
    if time.monotonic() > ready_deadline: raise RuntimeError('fixture receipt not available')
    time.sleep(0.2)
receipt = next(base.rglob('*-host.inspect.txt'))
pg_name = receipt.name.removesuffix('-host.inspect.txt') + '-pg'
out = Path('artifacts/part1-final-20261119/restart-db-snapshots.jsonl')
deadline = time.monotonic() + 420
with out.open('w') as stream:
    while time.monotonic() < deadline:
        result = subprocess.run(['docker','exec',pg_name,'psql','-XAt','-U','postgres','-c',sql],capture_output=True,text=True,timeout=15)
        row={'at':datetime.datetime.now(datetime.UTC).isoformat(),'exit_code':result.returncode}
        if result.returncode:
            row['diagnostic']='fixture container unavailable; no secrets collected'
            stream.write(json.dumps(row)+'\n'); stream.flush(); break
        row['state']=json.loads(result.stdout)
        stream.write(json.dumps(row)+'\n'); stream.flush()
        time.sleep(2)
print('Read-only scenario snapshots saved')
